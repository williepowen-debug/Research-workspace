<script>
    (function () {
      const container = document.getElementById('progress1');
      if (!container) return;

      const selector = 'slider1';
      if (!selector) return;

      const sliderEl = document.getElementById(selector);
      if (!sliderEl) return;

      const fillEl = container.querySelector('.js-progress-fill');
      const barEl  = container.querySelector('[role="progressbar"]');
      const srEl   = container.querySelector('.sr-only');

      function setProgress01(p) {
        p = Math.max(0, Math.min(1, Number.isFinite(p) ? p : 0));
        const pct = Math.round(p * 100);
        if (fillEl) fillEl.style.width = pct + '%';
        if (barEl)  barEl.setAttribute('aria-valuenow', String(pct));
        if (srEl)   srEl.textContent = pct + '%';
      }

      function countRealSlides(swiper) {
        const slides = swiper.slides || [];
        let total = 0;
        for (const s of slides) {
          if (!s.classList || !s.classList.contains('swiper-slide-duplicate')) total++;
        }
        return total || slides.length || 0;
      }

      function getVisibleCount(swiper) {
        let spv = swiper?.params?.slidesPerView;
        if (spv === 'auto' && typeof swiper.slidesPerViewDynamic === 'function') {
          spv = swiper.slidesPerViewDynamic('current', true);
        }
        const n = Number(spv);
        return Math.max(1, Number.isFinite(n) ? Math.round(n) : 1);
      }

      function computeProgress(swiper) {
        const total = countRealSlides(swiper);
        if (!total) return 1;
        const visible = getVisibleCount(swiper);
        let start = Number.isFinite(swiper.realIndex) ? swiper.realIndex : 0;
        if (swiper.params && swiper.params.centeredSlides) {
          start = Math.max(0, Math.min(total - visible, start - Math.floor(visible / 2)));
        }
        const end = Math.min(total, start + visible);
        return end / total;
      }

      function updateFromSwiper(swiper) {
        setProgress01(computeProgress(swiper));
      }

      function attach(swiper) {
        updateFromSwiper(swiper);
        swiper.on('slideChange', function () { updateFromSwiper(swiper); });
        swiper.on('progress', function () { updateFromSwiper(swiper); });
        swiper.on('resize', function () { updateFromSwiper(swiper); });
        swiper.on('observerUpdate', function () { updateFromSwiper(swiper); });
        swiper.on('update', function () { updateFromSwiper(swiper); });
      }

      function waitForSwiper() {
        if (sliderEl.swiper) { attach(sliderEl.swiper); return; }
        let tries = 0;
        const iv = setInterval(function () {
          tries++;
          if (sliderEl.swiper) { attach(sliderEl.swiper); clearInterval(iv); }
          if (tries > 40) { clearInterval(iv); setProgress01(1); }
        }, 100);
      }

      if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', waitForSwiper, { once: true });
      } else {
        waitForSwiper();
      }
    })();

    <!-- Simple cursor effect -->
    <script>
        (function () {
            try {
                var isTouch = 'ontouchstart' in window || navigator.maxTouchPoints > 0;
                if (isTouch) return;

                var dot = document.createElement('div');
                dot.id = 'cursor-dot';
                var size = 26; // px
                var style = dot.style;
                style.position = 'fixed';
                style.left = '0px';
                style.top = '0px';
                style.width = size + 'px';
                style.height = size + 'px';
                style.borderRadius = '50%';
                style.background = 'rgba(245, 245, 245, 0.9)'; // even lighter grey with opacity
                style.boxShadow = '0 0 12px rgba(245, 245, 245, 0.5)';
                style.pointerEvents = 'none';
                style.transform = 'translate(-50%, -50%) scale(1)';
                style.transition = 'transform 80ms ease-out, opacity 200ms ease';
                style.zIndex = '9999';
                style.opacity = '0';
                style.mixBlendMode = 'multiply';
                document.body.appendChild(dot);

                var lastX = 0, lastY = 0;
                var isDown = false;
                var isHoverImage = false;

                function updateTransform() {
                    var baseScale = isHoverImage ? 1.6 : 1;
                    var clickScale = isDown ? 0.85 : 1;
                    var scale = (baseScale * clickScale).toFixed(3);
                    style.transform = 'translate(-50%, -50%) scale(' + scale + ')';
                }
                document.addEventListener('mousemove', function (e) {
                    lastX = e.clientX; lastY = e.clientY;
                    style.left = lastX + 'px';
                    style.top = lastY + 'px';
                    style.opacity = '1';
                }, { passive: true });

                document.addEventListener('mouseleave', function () {
                    style.opacity = '0';
                }, { passive: true });

                document.addEventListener('mousedown', function () {
                    isDown = true;
                    updateTransform();
                }, { passive: true });

                document.addEventListener('mouseup', function () {
                    isDown = false;
                    updateTransform();
                }, { passive: true });

                function isImageElement(el) {
                    if (!el) return false;
                    if (el.closest && el.closest('img, picture, figure')) return true;
                    var tag = (el.tagName || '').toLowerCase();
                    return tag === 'img' || tag === 'picture';
                }

                document.addEventListener('mouseover', function (e) {
                    if (isImageElement(e.target)) {
                        isHoverImage = true;
                        updateTransform();
                    }
                }, { passive: true });

                document.addEventListener('mouseout', function (e) {
                    if (isImageElement(e.target)) {
                        isHoverImage = false;
                        updateTransform();
                    }
                }, { passive: true });
            } catch (_) { /* no-op */ }
        })();
    </script>
